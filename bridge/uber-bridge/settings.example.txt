# [uberGraph] Bridge settings, written by tools/bridge/bridge_settings.py. A missing file or key means OFF.
#
# ===================== RISKY OPERATIONS =====================
# Every option below changes the running game. An AI agent must ask the user and explain the consequences
# BEFORE turning one on, then use:  tools/bridge/bridge_settings.py set <key> true --user-approved "<the user's words>"
# (the tool refuses without an approval text, applies the change through tools/deploy and logs it).
# Turn an option off again when the analysis that needed it is done.
#
# allow_load=true lets the bridge load game assets by path.
# Consequences: the game thread can pause while the asset and its dependencies load; memory use goes up (this PC has hit its commit limit before); the asset's PostLoad logic runs; a strange asset could crash the game.
# Nothing is written to disk and the loaded objects are not kept by the bridge.
# Only /Game/ paths, one per request.
allow_load=false
#
# allow_hooks=true lets the bridge place an observe-only hook on a game function (watch_function).
# Consequences: the hook runs inside the game every time that function is called, so a frequently called function costs performance; hooks have been a crash suspect in earlier projects (a hook on every actor was never ruled out); the hook is removed when the game closes or with watch_function stop.
# It records strings only and changes nothing.
# At most 3 watchers at a time.
allow_hooks=false
#
# allow_console=true lets the bridge run one console command at a time (console_exec).
# Consequences: it changes the running game (a console variable, a stat overlay, a log category) until the game closes or the variable is set back; a wrong value can make the game slow, unstable or ugly.
# Only commands whose first word is stat, log, obj, showflag, dumpconsolecommands, listtimers or a cvar starting r. sg. t. run; no chaining; quit, open and travel commands are refused.
# The console's output text is not returned; read the effect with cvar_get or the UE4SS log.
allow_console=false
#
